## Приложение А — скрипт проверки цитат (для повторного прогона любой сессией)

```js
// check-registry-quotes.mjs — проверка дословности цитат researches/16 (EXP-0015).
// Запуск: node check-registry-quotes.mjs [путь к researches/16]
// Формат цитаты в реестре: строка «— Цитата», путь в бэктиках в квадратных скобках,
// двоеточие, затем дословная подстрока в ёлочках (см. regex ниже).
// Нормализация обеих сторон: BOM · \r\n→\n · NBSP→пробел · пробельные серии → один пробел.
import { readFileSync } from 'node:fs';
import { resolve, isAbsolute } from 'node:path';
const REG = process.argv[2] || 'researches/16_contour_requirements_registry.md';
const ROOT = resolve(REG, '..', '..');
const norm = (s) => s.replace(/^\uFEFF/, '').replace(/\r\n?/g, '\n')
  .replace(/\u00A0/g, ' ').replace(/\s+/g, ' ');
const reg = readFileSync(REG, 'utf8');
const rx = /Цитата\s*\[`([^`]+)`\]:\s*«([^«»]+)»/g;
const cache = new Map(); const miss = []; let total = 0;
for (const m of reg.matchAll(rx)) {
  total++;
  const p = isAbsolute(m[1]) ? m[1] : resolve(ROOT, m[1]);
  if (!cache.has(p)) { try { cache.set(p, norm(readFileSync(p, 'utf8'))); } catch { cache.set(p, null); } }
  const hay = cache.get(p);
  if (!hay) { miss.push(`НЕТ ФАЙЛА: ${m[1]}`); continue; }
  if (!hay.includes(norm(m[2]))) miss.push(`НЕ НАЙДЕНО в ${m[1]}: «${m[2].slice(0, 60)}…»`);
}
const count = (pfx) => (reg.match(new RegExp(`^- \\*\\*${pfx}\\d+`, 'gm')) || []).length;
console.log(`Цитат проверено: ${total}; промахов: ${miss.length}`);
for (const x of miss) console.log('  🔴 ' + x);
console.log(`Позиций: I=${count('I')} M=${count('M')} P=${count('P')} G=${count('G')} T=${count('T')} C=${count('C')} QA=${count('QA')} O=${count('O')}; DEF-строк=${(reg.match(/^\| DEF\d+/gm) || []).length}`);
process.exit(miss.length ? 1 : 0);
```
